\begin{vmatrix} \textcolor{red}{1} & \textcolor{red}{1} & \textcolor{red}{0} \\ \textcolor{red}{0} & \textcolor{red}{1} & \textcolor{red}{1} \\ \textcolor{red}{0} & \textcolor{red}{1} & \textcolor{red}{0} \end{vmatrix}
$$

Sviluppo secondo la prima riga

$$
\begin{vmatrix} \textcolor{blue}{1} & \textcolor{red}{1} & \textcolor{red}{0} \\ \textcolor{blue}{0} & \textcolor{red}{1} & \textcolor{red}{1} \\ \textcolor{blue}{0} & \textcolor{red}{1} & \textcolor{red}{0} \end{vmatrix} = \textcolor{blue}{1} \cdot \begin{vmatrix} \textcolor{red}{1} & \textcolor{red}{1} \\ \textcolor{red}{1} & \textcolor{red}{0} \end{vmatrix} = \textcolor{blue}{1} \cdot (-1) = -1
$$
