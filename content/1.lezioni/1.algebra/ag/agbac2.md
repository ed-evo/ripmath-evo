$$

$$
\textcolor{blue}{3x - 2x - 4x \le -8 - 2}
$$

sommo

$$
\textcolor{blue}{-3x \le -10}
$$

$$
\textcolor{blue}{\frac{-3x}{-3} \ge \frac{-10}{-3}}
$$

semplifico

$$
\textcolor{blue}{x \ge 10/3}
$$
