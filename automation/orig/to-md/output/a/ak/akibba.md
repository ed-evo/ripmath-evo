x - 1 \geq 0 \\
(x - 1)^2 \leq x^2 - 4
\end{cases}
}
$$

sviluppiamo la seconda equazione

$$
\textcolor{blue}{
\begin{cases}
x - 1 \geq 0 \\
x^2 - 2x + 1 \leq x^2 - 4
\end{cases}
}
$$

porto tutti i termini prima del disuguale

$$
\textcolor{blue}{
\begin{cases}
x - 1 \geq 0 \\
x^2 - 2x + 1 - x^2 + 4 \leq 0
\end{cases}
}
$$

sommo i termini simili

$$
\textcolor{blue}{
\begin{cases}
x + 3 \geq 0 \\
- 2x + 5 \leq 0
\end{cases}
}
$$

porto il termine noto dopo l'uguale

$$
\textcolor{blue}{
\begin{cases}
x + 3 \geq 0 \\
- 2x \leq -5
\end{cases}
}
$$

cambio di segno e di verso

$$
\textcolor{blue}{
\begin{cases}
x - 1 \geq 0 \\
2x \geq 5
\end{cases}
}
$$

$$
\textcolor{blue}{
\begin{cases}
