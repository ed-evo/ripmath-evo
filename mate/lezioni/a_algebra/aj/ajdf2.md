# Calcoli per i complementi algebrici degli elementi del determinante

$$
\textcolor{red}{\begin{vmatrix} 1 & 2 & 1 \\ 1 & 1 & -2 \\ 2 & 2 & 1 \end{vmatrix}}
$$

$$
\textcolor{red}{C_{1,1} = + \begin{vmatrix} 1 & -2 \\ 2 & 1 \end{vmatrix} = 1 \cdot 1 - (-2) \cdot 2 = 1 + 4 = 5}
$$

$$
\textcolor{red}{C_{1,2} = - \begin{vmatrix} 1 & -2 \\ 2 & 1 \end{vmatrix} = -[ 1 \cdot 1 - (-2) \cdot 2] = -(1 + 4) = -5}
$$

$$
\textcolor{red}{C_{1,3} = + \begin{vmatrix} 1 & 1 \\ 2 & 2 \end{vmatrix} = 1 \cdot 2 - 1 \cdot 2 = 1 \cdot (0) = 0}
$$

$$
\textcolor{red}{C_{2,1} = - \begin{vmatrix} 2 & 1 \\ 2 & 1 \end{vmatrix} = -(2 \cdot 1 - 2 \cdot 1) = -(2 - 2) = 0}
$$

$$
\textcolor{red}{C_{2,2} = + \begin{vmatrix} 1 & 1 \\ 2 & 1 \end{vmatrix} = 1 \cdot 1 - 1 \cdot 2 = 1 - 2 = -1}
$$

$$
\textcolor{red}{C_{2,3} = - \begin{vmatrix} 1 & 2 \\ 2 & 2 \end{vmatrix} = -(1 \cdot 2 - 2 \cdot 2) = -(2 - 4) = 2}
$$

$$
\textcolor{red}{C_{3,1} = + \begin{vmatrix} 2 & 1 \\ 1 & -2 \end{vmatrix} = 2 \cdot (-2) - 1 \cdot 1 = -4 - 1 = -5}
$$

$$
\textcolor{red}{C_{3,2} = - \begin{vmatrix} 1 & 1 \\ 1 & -2 \end{vmatrix} = -[1 \cdot (-2) - 1 \cdot 1] = -(-2 - 1) = 3}
$$

$$
\textcolor{red}{C_{3,3} = + \begin{vmatrix} 1 & 2 \\ 1 & 1 \end{vmatrix} = 1 \cdot 1 - 2 \cdot 1 = 1 - 2 = -1}
$$
